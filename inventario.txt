<script>

let productos=[]
let ventas=0
let gastos=0
let ganancia=0

function actualizarTabla(){

let tabla=document.getElementById("tablaProductos")
tabla.innerHTML=""

productos.forEach((p,i)=>{

tabla.innerHTML+=`
<tr>
<td>${p.nombre}</td>
<td>${p.stock}</td>
<td><button onclick="vender(${i})">Vender</button></td>
</tr>
`

})

document.getElementById("ventas").innerText=ventas
document.getElementById("gastos").innerText=gastos
document.getElementById("ganancia").innerText=ganancia

}

function agregarProducto(){

let nombre=document.getElementById("nombre").value
let compra=parseFloat(document.getElementById("compra").value)
let venta=parseFloat(document.getElementById("venta").value)
let cantidad=parseInt(document.getElementById("cantidad").value)

productos.push({
nombre:nombre,
compra:compra,
venta:venta,
stock:cantidad
})

gastos+=compra*cantidad

actualizarTabla()
actualizarGrafico()

}

function vender(i){

let p=productos[i]

if(p.stock>0){

p.stock--

ventas+=p.venta
ganancia+=p.venta-p.compra

actualizarTabla()
actualizarGrafico()

}

}

let chart

function actualizarGrafico(){

let data={
labels:["Ventas","Gastos","Ganancia"],
datasets:[{
label:"Dinero",
data:[ventas,gastos,ganancia],
backgroundColor:["green","red","blue"]
}]
}

if(chart){
chart.destroy()
}

chart=new Chart(document.getElementById("grafico"),{
type:"bar",
data:data
})

}
